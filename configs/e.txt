#!/usr/bin/env python3
import argparse, csv, datetime, os, re, yaml
from netmiko import ConnectHandler

def cargar(ruta):
    with open(ruta) as f:
        return yaml.safe_load(f)

def seccion(texto, nombre):
    salida, dentro = [], False
    for linea in texto.splitlines():
        if linea.startswith(nombre):
            dentro = True
            salida.append(linea)
        elif dentro and linea.startswith(" "):
            salida.append(linea)
        else:
            dentro = False
    return "\n".join(salida)

def cumple(regla, texto):
    if "seccion" in regla:
        texto = seccion(texto, regla["seccion"])
    hallado = re.search(regla["regex"], texto, re.M) is not None
    return (not hallado) if regla["tipo"] == "ausente" else hallado

def aplicables(reglas, dev):
    return [r for r in reglas if "solo" not in r or dev["nombre"] in r["solo"]]

def evaluar(conn, reglas):
    cache, res = {}, {}
    for r in reglas:
        if r["cmd"] not in cache:
            cache[r["cmd"]] = conn.send_command(r["cmd"], read_timeout=90)
        res[r["id"]] = cumple(r, cache[r["cmd"]])
    return res

def guardar_config(conn, dev, salida, etiqueta):
    cmd = "show running-config" if dev["device_type"] == "cisco_ios" else "show full-configuration"
    os.makedirs(os.path.join(salida, "configs"), exist_ok=True)
    with open(os.path.join(salida, "configs", f"{dev['nombre']}_{etiqueta}.txt"), "w") as f:
        f.write(conn.send_command(cmd, read_timeout=90))

def procesar(dev, reglas, salida, aplicar):
    conn = ConnectHandler(device_type=dev["device_type"], host=dev["host"],
                          username=dev["username"], password=dev["password"],
                          conn_timeout=30)
    reglas = aplicables(reglas, dev)
    guardar_config(conn, dev, salida, "antes")
    antes = evaluar(conn, reglas)
    despues = {}
    if aplicar:
        for r in reglas:
            if not antes[r["id"]] and r.get("fix"):
                conn.send_config_set(r["fix"])
        if dev["device_type"] == "cisco_ios":
            conn.save_config()
        guardar_config(conn, dev, salida, "despues")
        despues = evaluar(conn, reglas)
    conn.disconnect()
    filas = []
    for r in reglas:
        a = "CUMPLE" if antes[r["id"]] else "NO CUMPLE"
        if aplicar:
            d = "CUMPLE" if despues[r["id"]] else "NO CUMPLE"
            accion = "Aplicada" if (not antes[r["id"]] and despues[r["id"]]) else \
                     ("Sin cambio" if antes[r["id"]] else "FALLO al aplicar")
        else:
            d, accion = "-", "-"
        filas.append((dev["nombre"], r["id"], r["titulo"], r.get("cis", ""), a, d, accion))
    return filas

def main():
    p = argparse.ArgumentParser(description="Auditor de hardening (CIS)")
    p.add_argument("--modo", choices=["audit", "apply"], default="audit")
    p.add_argument("--dispositivos", default="dispositivos.yaml")
    p.add_argument("--reglas", default="reglas.yaml")
    p.add_argument("--salida", default="salida")
    a = p.parse_args()

    reglas = cargar(a.reglas)
    todos = []
    for dev in cargar(a.dispositivos):
        try:
            todos += procesar(dev, reglas[dev["device_type"]], a.salida, a.modo == "apply")
        except Exception as e:
            todos.append((dev["nombre"], "-", "Error de conexion", "", f"ERROR: {e}", "-", "-"))

    os.makedirs(a.salida, exist_ok=True)
    marca = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    base = os.path.join(a.salida, f"informe_{a.modo}_{marca}")
    cab = ["Dispositivo", "ID", "Recomendacion", "CIS", "Antes", "Despues", "Accion"]
    with open(base + ".csv", "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(cab)
        w.writerows(todos)
    with open(base + ".md", "w") as f:
        f.write(f"# Informe de hardening (modo {a.modo})\n\n")
        f.write("| " + " | ".join(cab) + " |\n|" + "---|" * len(cab) + "\n")
        for fila in todos:
            f.write("| " + " | ".join(fila) + " |\n")
    for d in sorted({x[0] for x in todos}):
        filas = [x for x in todos if x[0] == d and x[1] != "-"]
        ok = sum(1 for x in filas if x[4] == "CUMPLE")
        linea = f"{d}: {ok}/{len(filas)} cumplen"
        if a.modo == "apply":
            ok2 = sum(1 for x in filas if x[5] == "CUMPLE")
            linea += f" -> {ok2}/{len(filas)} tras aplicar"
        print(linea)
    print("Informe:", base + ".md")

if __name__ == "__main__":
    main()
